第一題:米字圖形

def print_filled_circle_in_grid():
    # 圖形的大小
    width, height = 10, 10
    # 圓心的座標和半徑
    cx, cy = 5, 5  # 圓心 (5, 5)
    r = 4  # 半徑

    # 繪製每個點
    for y in range(height):
        for x in range(width):
            # 計算 (x, y) 到圓心的距離
            distance = (x - cx)**2 + (y - cy)**2
            if distance <= r**2:  # 判斷是否在圓內
                print("*", end="")
            else:
                print(" ", end="")
        print()  # 換行

print_filled_circle_in_grid()

第二題:網頁上的方格與塗色

from browser import html
from browser import document as doc

# 建立畫布
canvas = html.CANVAS(width=200, height=200)  # 200x200的畫布大小
brython_div = doc["brython_div1"]
brython_div <= canvas

# 獲取畫布的繪圖上下文
ctx = canvas.getContext("2d")

# 每個格子的像素大小
gs = 20  # 每格20像素
tc = 10  # 10x10的格子數量

# 繪製矩形的功能
def dRect(lux, luy, w, h, s=1, c='#ff0000'):
    ctx.lineWidth = s
    ctx.strokeStyle = c
    ctx.beginPath()
    ctx.rect(lux, luy, w, h)
    ctx.stroke()

# 繪製網格
def grid(width, height, grid_pix):
    for i in range(width):
        for j in range(height):
            dRect(i * grid_pix, j * grid_pix, grid_pix, grid_pix, 1, "lightgrey")

# 填充圓形區域
def fill(x, y, color):
    ctx.fillStyle = color
    ctx.fillRect(x - 1, y - 1, gs, gs)

# 在網格中填充圓形
def draw_circle_in_grid(cx, cy, r):
    for i in range(tc):
        for j in range(tc):
            # 計算每個格子中心的座標
            x = i * gs + gs // 2
            y = j * gs + gs // 2
            # 計算該格子是否在圓形內部
            distance = (x - cx) ** 2 + (y - cy) ** 2
            if distance <= r ** 2:
                fill(i * gs, j * gs, "black")

# 繪製網格並在網格內繪製實心圓形
grid(tc, tc, gs)

# 圓心在(5, 5)，半徑4個格子，計算圓心在畫布上的位置
center_x = 5 * gs + gs // 2
center_y = 5 * gs + gs // 2
radius = 4 * gs  # 半徑為4格

# 在網格內繪製實心圓形
draw_circle_in_grid(center_x, center_y, radius)
